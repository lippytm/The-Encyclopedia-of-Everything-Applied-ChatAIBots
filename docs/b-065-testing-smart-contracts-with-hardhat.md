### 10 Common Errors

1. **revert** — Transaction failed. Fix: Check require() conditions.
2. **out of gas** — Execution ran out of gas. Fix: Increase gas limit or optimize.
3. **nonce too low** — Transaction already processed. Fix: Increment nonce.
4. **invalid opcode** — Contract bug. Fix: Check for zero-address or overflow.
5. **execution reverted** — Explicit revert() called. Fix: Read the revert reason.
6. **insufficient funds** — Not enough ETH for gas. Fix: Fund the wallet.
7. **contract not deployed** — Wrong address. Fix: Verify contract address on explorer.
8. **ABI mismatch** — Wrong ABI for contract. Fix: Recompile and export fresh ABI.
9. **stack too deep** — Too many local variables. Fix: Use structs or split functions.
10. **reentrancy** — External call before state update. Fix: Use CEI pattern.

const { expect } = require("chai");
const { ethers } = require("hardhat");

// BCL-L0-B065-ContractTester capstone test suite
describe("HelloWorld", function () {
  async function deployFixture() {
    const [owner, user] = await ethers.getSigners();
    const HelloWorld = await ethers.getContractFactory("HelloWorld");
    const contract = await HelloWorld.deploy("Hello, ACSS!");
    return { contract, owner, user };
  }

  it("Returns greeting", async function () {
    const { contract } = await deployFixture();
    expect(await contract.greeting()).to.equal("Hello, ACSS!");
  });

  it("Increments greetCount on greet()", async function () {
    const { contract } = await deployFixture();
    await contract.greet();
    expect(await contract.greetCount()).to.equal(1n);
  });

  it("Only owner can setGreeting", async function () {
    const { contract, user } = await deployFixture();
    await expect(contract.connect(user).setGreeting("hack")).to.be.revertedWith("Not owner");
  });
});
